      for (const file of files.filter((entry) => !entry.source && entry.patch)) {
        const matches = parsePatchFiles(file.patch, { throwOnError: true }).flatMap((patch) => patch.files).filter((entry) => entry.name === file.path);
        require(matches.length === 1, `Patch needs exactly one matching file: ${file.path}`);
        const diff = matches[0];
        const types = { added: ["new"], removed: ["deleted"], renamed: ["rename-pure", "rename-changed"], modified: ["change"], changed: ["change"], copied: ["new", "change", "rename-pure", "rename-changed"] };
        require(types[file.status].includes(diff.type) && (diff.prevName || file.path) === (file.previousPath || file.path), `Patch paths/status do not match file metadata: ${file.path}`);
        for (const excerpt of stops.flatMap((stop) => stop.excerpts).filter((entry) => entry.path === file.path)) {
          const side = excerpt.side === "old" ? "deletion" : "addition";
          require(diff.hunks.some((hunk) => hunk[`${side}Count`] > 0 && excerpt.start >= hunk[`${side}Start`] && excerpt.end < hunk[`${side}Start`] + hunk[`${side}Count`]), `Excerpt is outside supplied ${excerpt.side} patch hunks: ${file.path}:${excerpt.start}–${excerpt.end}`);
        }
        patchDiffs.set(file.path, diff);
      }
    }

    function selectedCode(card) {
      const range = card.selection;
      if (!range) return null;
      const side = range.side || (card.excerpt.side === "old" ? "deletions" : "additions");
      require(["deletions", "additions"].includes(side) && (!range.endSide || range.endSide === side), "Select lines from just one side to copy.");
      require(Number.isSafeInteger(range.start) && Number.isSafeInteger(range.end) && range.start > 0 && range.end > 0, "Select a valid line range to copy.");
      const start = Math.min(range.start, range.end), end = Math.max(range.start, range.end);
      const name = side === "deletions" ? "old" : "new";
      let contents;
      if (card.file.source) {
        const source = card.file.source[name === "old" ? "old" : "next"];
        const lines = source === null ? [] : source.match(/[^\n]*\n|[^\n]+$/g) || [];
        require(end <= lines.length, "Selected source lines are unavailable.");
        contents = lines.slice(start - 1, end).join("");
      } else {
        const diff = patchDiffs.get(card.file.path), parts = [];
        require(diff, "Selected source lines are unavailable.");
        const column = name === "old" ? "deletion" : "addition";
        let cursor = start;
        for (const hunk of diff.hunks) {
          const first = hunk[`${column}Start`], limit = first + hunk[`${column}Count`];
          if (cursor < first) break;
          if (cursor >= limit) continue;
          const count = Math.min(end + 1, limit) - cursor;
          const index = hunk[`${column}LineIndex`] + cursor - first;
          parts.push(diff[`${column}Lines`].slice(index, index + count).join(""));
          cursor += count;
          if (cursor > end) break;
        }
        require(cursor > end, "Selected source lines are not all present in this patch.");
        contents = parts.join("");
      }
      return { side: name, start, end, contents };
    }

      persistPreferences();
